cmake_minimum_required(VERSION 3.16)
project(aiglass_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LVGL_DIR "${WORKSPACE}/.host-debug/deps/lvgl" CACHE PATH "LVGL 8.3.11 source")
set(CURL_HEADERS "${WORKSPACE}/.host-debug/deps/curl-include" CACHE PATH "libcurl headers")
find_library(CURL_LIB NAMES curl libcurl.so.4 REQUIRED)
find_package(Threads REQUIRED)
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_DIR}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${LVGL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
set(UI "${CMAKE_CURRENT_SOURCE_DIR}/../lvgl-home")
add_executable(aiglass-host main.c media_unavailable.c
    "${UI}/aiglass_home.c" "${UI}/aiglass_game.c" "${UI}/aiglass_services.c"
    "${UI}/aiglass_pairing.c" "${UI}/aiglass_phone.c" "${UI}/aiglass_http.c" "${UI}/aiglass_media.c"
    "${UI}/fonts/aiglass_font_16.c" "${UI}/third_party/cJSON.c")
target_include_directories(aiglass-host PRIVATE "${UI}" "${UI}/../mpp-media" "${CURL_HEADERS}")
target_compile_options(aiglass-host PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
target_link_libraries(aiglass-host PRIVATE lvgl ${CURL_LIB} Threads::Threads m)
option(AG_SANITIZE "Address/undefined sanitizer for host debugging" OFF)
if(AG_SANITIZE)
    foreach(target lvgl aiglass-host)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(aiglass-host PRIVATE -fsanitize=address,undefined)
endif()
